འོད་གསལ་རྒྱལ་པོ་ངོ་སྤྲོད་པས། །
ངོ་སྤྲོད་རྒྱལ་པོའང་འདི་ཡིན་ནོ། །
༼༡༽འོད་གསལ་ཆོས་སྐུར་༼༢༽རྫོགས་སངས་རྒྱས། །༼༣༽རང་བྱུང་ཡེ་ཤེས་༼༤༽ལྷན་སྐྱེས་བདེ། །༼༥༽དང་པོའི་སངས་རྒྱས་༼༦༽དུས་འཁོར་ལོ། །༼༧༽དགྱེས་རྡོར་༼༨༽བདེ་མཆོག་༼༩༽གསང་འདུས་དང༌། །༼༡༠༽སྒྱུ་འཕྲུལ་དྲྭ་བ་༼༡༡༽ཆོས་ཀྱི་དབྱིངས། །༼༡༢༽རྡོ་རྗེ་སེམས་དཔའ་༼༡༣༽ཨེ་ཝཾ་སོགས། །

ཐམས་ཅད་བདེ་ལེགས་སྙིང་པོའི་དོན། །

༣༽ཤློཀའི་གྲངས་སྨོས་པ་ནི།
དོན་ཚན་བཅུ་དྲུག་ཡོད་དོ། །
༤༽ཤིས་པ་བརྗོད་པ་ནི།
མངྒལཾ། །
(མ་དཔེ་ནི་འབྲས་གནས་བཅུའི་དཔེ་རྙིང་ཕྱི་ཀ ༢༠༤ ཡིན།)
ཞེས་པའང་གནས་མཆོག་ཝཱ་རཱ་ཎ་སཱིར་ཁྲ་འགུ་བཛྲ་བིདྱཱ་གཙུག་ལག་སློབ་གཉེར་ཁང་དུ། ཕྱི་ལོ་༢༠༢༠ ཟླ་༩ སྤྱི་ཚེས་༢༤ དང་བོད་ཚེས་༨ ཉིན་མོའི་དུས་བཟང་དང་སྟབས་བསྟུན་ཏེ་ཐུན་མཚམས་སུ་གུས་དོལ་པོ་པ་གཞན་སྟོང་འཇིགས་མེད་གུ་ཡངས་ནས། ༧ཇོ་ནང་ཀུན་མཁྱེན་དོལ་པོ་སངས་རྒྱས་ཀྱིས་མཛད་པའི་གསོལ་འདེབས་ཀྱི་རྒྱལ་པོ་ཞེས་བྱ་བའི་གཞུང་བཟང་འདི་ཉིད་ལ་ས་བཅད་སྟབས་བདེ་ཞིག་ཁ་སྣོན་བགྱིས་པའི་དགེ་རྩའི་མཐུས། རྫོགས་ལྡན་ལུགས་ཀྱི་བདེ་གཤེགས་སྙིང་པོའི་ཆོས་ཚུལ་རྣམས་མ་ནོར་བར་ཇི་བཞིན་ཡོངས་སུ་རྫོགས་པར་བདག་གཞན་ཀུན་གྱིས་རྟོགས་ཤིང་། མཐར་གཞན་སྟོང་ཨེ་ཝཾ་མ་ནིང་གི་གོ་འཕང་ཆེན་པོ་འཐོབ་པར་སྨོན་ནོ།། །།
